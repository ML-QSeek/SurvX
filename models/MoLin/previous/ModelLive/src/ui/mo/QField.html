⚙️ QFIELD 配置示例 · YAML 声明式 v0.1 · 规范草案 
# EGO 七层配置示例
ego:
  id: "example_agent"
  structure: ["模块A", "模块B"]
  compute: "输入 → 处理 → 输出"
  relations: ["与Field A关联", "依赖Field B"]
  constraints: "边界条件定义"
  features: { "状态1": "值" }
  goals: ["目标1", "目标2"]
  symbol: { name: "example", interface: "protocol" }

# Field 五层配置示例
field:
  id: "example_field"
  structure: ["组件X", "组件Y"]
  compute: "演化规则"
  relations: ["内部连接"]
  constraints: "物理边界"
  features: { "环境参数": "值" }

# 解释器驱动
interpreter:
  tick: "动态"
  mode: "声明式执行"
  validation: "约束检查 · 结构验证"
